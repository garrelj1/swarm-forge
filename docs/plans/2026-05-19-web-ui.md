# SwarmForge Web UI Implementation Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Build a Rust/Axum web server that streams agent terminal output and inter-agent messages to a browser dashboard, with per-agent keystroke relay for responding to permission prompts.

**Architecture:** `swarmforge-ui` is a self-contained Rust binary. `swarmforge.sh` adds `tmux pipe-pane` per agent (writing to `.swarmforge/panes/<role>.log`) and auto-builds/launches the binary after agents start. The server reads `sessions.tsv` to discover roles, tails pane logs and `logs/agent_messages.log`, and streams both to the browser via SSE. A plain HTML/xterm.js frontend renders one terminal panel per agent with a keystroke input. Sending keys POSTs to `/send/:role` which runs `tmux send-keys`.

**Tech Stack:** Rust, Axum 0.7, Tokio (full), serde_json, clap 4, xterm.js 5 (CDN, no build step)

```rust
        if cols.len() >= 3 {
            Some(Session { role: cols[1].to_string(), session: cols[2].to_string() })

        }
```

```rust
pub fn parse_message_line(line: &str) -> Option<(String, String)> {
    // Format: [timestamp] [swarmforge-role] message text
    let after_ts = line.find("] [")?;
    let rest = &line[after_ts + 3..];
    let session_end = rest.find(']')?;
    let session = &rest[..session_end];
    let text = rest[session_end + 2..].trim().to_string();
    let role = session.strip_prefix("swarmforge-").unwrap_or(session).to_string();
    Some((role, text))
}
```

```rust
use std::path::PathBuf;
use std::time::Duration;
use tokio::io::{AsyncBufReadExt, BufReader};
use tokio::sync::mpsc;
use tokio::time;

pub async fn tail(path: PathBuf, tx: mpsc::Sender<String>) {
    // Wait for file to exist (agents may start slightly after UI)
    let deadline = time::Instant::now() + Duration::from_secs(10);
    while !path.exists() {
        if time::Instant::now() > deadline { return; }
        time::sleep(Duration::from_millis(500)).await;
    }

    let file = match tokio::fs::File::open(&path).await {
        Ok(f) => f,
        Err(_) => return,
    };
    let mut reader = BufReader::new(file);
    let mut line = String::new();

    loop {
        line.clear();
        match reader.read_line(&mut line).await {
            Ok(0) => time::sleep(Duration::from_millis(200)).await,
            Ok(_) => { if tx.send(line.clone()).await.is_err() { return; } }
            Err(_) => return,
        }
    }
}
```

```rust
async fn sse_handler(
    State(state): State<AppState>,
) -> Sse<impl tokio_stream::Stream<Item = Result<Event, Infallible>>> {
    let (tx, rx) = mpsc::channel::<events::Payload>(256);

    // Send roles event immediately
    let roles: Vec<String> = state.roles.iter().map(|s| s.role.clone()).collect();
    let tx2 = tx.clone();
    tokio::spawn(async move { let _ = tx2.send(events::Payload::Roles { roles }).await; });

    // Tail each pane log
    for session in state.roles.iter() {
        let log = state.working_dir.join(format!(".swarmforge/panes/{}.log", session.role));
        let role = session.role.clone();
        let tx2 = tx.clone();
        tokio::spawn(async move {
            let (ltx, mut lrx) = mpsc::channel(256);
            tokio::spawn(tailer::tail(log, ltx));
            while let Some(text) = lrx.recv().await {
                if tx2.send(events::Payload::Pane { role: role.clone(), text }).await.is_err() { break; }
            }
        });
    }

    // Tail message log
    {
        let log = state.working_dir.join("logs/agent_messages.log");
        let tx2 = tx.clone();
        tokio::spawn(async move {
            let (ltx, mut lrx) = mpsc::channel(256);
            tokio::spawn(tailer::tail(log, ltx));
            while let Some(line) = lrx.recv().await {
                if let Some((role, text)) = events::parse_message_line(&line) {
                    if tx2.send(events::Payload::Message { role, text }).await.is_err() { break; }
                }
            }
        });
    }

    let stream = ReceiverStream::new(rx).map(|payload| {
        let name = events::event_name(&payload);
        let data = serde_json::to_string(&payload).unwrap();
        Ok::<Event, Infallible>(Event::default().event(name).data(data))
    });

    Sse::new(stream).keep_alive(axum::response::sse::KeepAlive::default())
}
```

```rust
#[derive(serde::Deserialize)]
struct SendBody { keys: String }

async fn send_handler(
    Path(role): Path<String>,
    State(state): State<AppState>,
    Json(body): Json<SendBody>,
) -> axum::http::StatusCode {
    let Some(session) = state.roles.iter().find(|s| s.role == role).map(|s| s.session.clone()) else {
        return axum::http::StatusCode::NOT_FOUND;
    };
    let target = tmux_target(&session);
    if !body.keys.is_empty() {
        std::process::Command::new("tmux")
            .args(["send-keys", "-t", &target, "-l", "--", &body.keys])
            .output().ok();
    }
    std::process::Command::new("tmux")
        .args(["send-keys", "-t", &target, "Enter"])
        .output().ok();
    axum::http::StatusCode::OK
}
```
